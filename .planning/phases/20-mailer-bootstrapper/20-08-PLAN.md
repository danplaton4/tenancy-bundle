  <action>
    Create `src/Command/Install/Step/MailerSetupStep.php`. Mirror `BundlesPhpInstaller`'s structural pattern exactly. Key design — the step has THREE sub-operations executed in order: (a) `updateEntity()`, (b) `scaffoldMigration()`, (c) `updateTenancyYaml()`. Each sub-operation returns its own status surfaced via `$io`; the parent `run()` returns the entity result as the primary status (migration + yaml are informational).

    ```php
    <?php
    declare(strict_types=1);

    namespace Tenancy\Bundle\Command\Install\Step;

    use PhpParser\Node;
    use PhpParser\NodeFinder;
    use PhpParser\ParserFactory;
    use PhpParser\PrettyPrinter\Standard;
    use Symfony\Component\Console\Style\SymfonyStyle;
    use Symfony\Component\Filesystem\Filesystem;
    use Tenancy\Bundle\Command\Install\InstallResult;
    use Tenancy\Bundle\Mailer\TenantMailerConfigTrait;

    /**
     * tenancy:install --with-mailer sub-step.
     *
     * Implements all three D-09 sub-actions:
     *   1. Locates the user's Tenant entity (from the `tenancy.tenant_entity_class`
     *      config or via fallback path) and AST-inserts `use TenantMailerConfigTrait;`.
     *      On non-standard layouts → refuses with manual snippet (DEC-INST-02 pattern).
     *   2. Scaffolds the Doctrine migration file (or prints the SQL snippet when
     *      doctrine/migrations is absent).
     *   3. Appends a commented-out `mailer:` defaults block to
     *      `config/packages/tenancy.yaml` (idempotent — no-op when section already
     *      present; prints snippet when file is missing).
     *
     * All file writes are atomic with a .bak; on php -l failure post-mutation, the
     * .bak is restored.
     */
    final class MailerSetupStep
    {
        /**
         * The commented-out defaults block appended to config/packages/tenancy.yaml.
         * Lines begin with `#` so the defaults are visible but inert (D-09 sub-action 3).
         */
        public const TENANCY_YAML_MAILER_BLOCK = <<<'YAML'

# Per-tenant Mailer defaults (BOOT-04). Uncomment + tune to enable.
# mailer:
#     strategy: x_transport
#     transport_cache_size: 32
#     sanitize_exceptions: true
YAML;

        public function __construct(
            private readonly Filesystem $filesystem = new Filesystem(),
            private readonly ?\Closure $lintRunner = null, // for testing
        ) {}

        public function run(
            SymfonyStyle $io,
            string $tenantEntityPath,
            string $migrationsDir,
            string $tenancyYamlPath,
            bool $dryRun = false,
        ): InstallResult {
            if (!class_exists(ParserFactory::class)) {
                $io->note('nikic/php-parser not installed — printing manual instructions:');
                $io->listing([
                    'Add `use \\Tenancy\\Bundle\\Mailer\\TenantMailerConfigTrait;` to your Tenant entity class body.',
                    'Run `bin/console doctrine:migrations:diff` to generate the column migration.',
                    'Append the following to config/packages/tenancy.yaml under the tenancy: root:'.self::TENANCY_YAML_MAILER_BLOCK,
                ]);
                return InstallResult::devDependencyMissing();
            }

            // 1. AST-detect + insert trait (load-bearing — return value drives parent status)
            $entityResult = $this->updateEntity($tenantEntityPath, $dryRun, $io);
            // 2. Scaffold migration (informational — surfaced via $io)
            $this->scaffoldMigration($migrationsDir, $dryRun, $io);
            // 3. Append commented-out mailer block to tenancy.yaml (informational — surfaced via $io)
            $this->updateTenancyYaml($tenancyYamlPath, $dryRun, $io);

            return $entityResult;
        }

        private function updateEntity(string $path, bool $dryRun, SymfonyStyle $io): InstallResult
        {
            if (!is_file($path)) {
                $io->warning(sprintf('Tenant entity not found at "%s" — skipping.', $path));
                return InstallResult::refusedNonStandard(/* manual snippet */ '...');
            }

            $code = (string) file_get_contents($path);
            $parser = (new ParserFactory())->createForNewestSupportedVersion();
            try {
                $ast = $parser->parse($code);
            } catch (\Throwable $e) {
                return InstallResult::refusedNonStandard('Could not parse '.$path.': '.$e->getMessage());
            }

            $finder = new NodeFinder();
            $classes = $finder->findInstanceOf($ast, Node\Stmt\Class_::class);
            if (count($classes) !== 1) {
                return InstallResult::refusedNonStandard('Expected exactly one class in '.$path.'; got '.count($classes).'.');
            }
            /** @var Node\Stmt\Class_ $class */
            $class = $classes[0];

            // Check for existing TraitUse
            foreach ($class->stmts as $stmt) {
                if ($stmt instanceof Node\Stmt\TraitUse) {
                    foreach ($stmt->traits as $traitName) {
                        $name = $traitName->toString();
                        if (str_ends_with($name, 'TenantMailerConfigTrait')) {
                            return InstallResult::alreadyInstalled();
                        }
                    }
                }
            }

            // Insert as first statement in class body
            $useStmt = new Node\Stmt\TraitUse([new Node\Name\FullyQualified(TenantMailerConfigTrait::class)]);
            array_unshift($class->stmts, $useStmt);

            $printer = new Standard();
            $newCode = $printer->prettyPrintFile($ast);

            if ($dryRun) {
                $io->section('--with-mailer (dry-run): proposed Tenant entity mutation');
                $io->text('A `use \\Tenancy\\Bundle\\Mailer\\TenantMailerConfigTrait;` statement would be inserted as the first statement in the class body.');
                return InstallResult::ok('dry-run: would mutate '.$path);
            }

            // Atomic write with .bak
            $bakPath = $path.'.bak.'.date('YmdHis');
            $this->filesystem->copy($path, $bakPath);
            $this->filesystem->dumpFile($path, $newCode);

            // Post-mutation lint check
            $linter = $this->lintRunner ?? static function (string $p): int {
                exec(escapeshellcmd(PHP_BINARY).' -l '.escapeshellarg($p).' 2>&1', $out, $exit);
                return $exit;
            };
            if (0 !== $linter($path)) {
                $this->filesystem->copy($bakPath, $path);
                return InstallResult::refusedNonStandard('php -l failed on '.$path.' after mutation; original restored from '.$bakPath);
            }

            $io->success('Inserted `use TenantMailerConfigTrait;` into '.$path.' (backup: '.$bakPath.')');
            return InstallResult::ok('Inserted trait into '.$path);
        }

        private function scaffoldMigration(string $migrationsDir, bool $dryRun, SymfonyStyle $io): InstallResult
        {
            // Build SQL up/down statements
            $up = 'ALTER TABLE tenancy_tenants '
                .'ADD COLUMN mailer_dsn VARCHAR(255) DEFAULT NULL, '
                .'ADD COLUMN mailer_from VARCHAR(255) DEFAULT NULL, '
                .'ADD COLUMN mailer_reply_to VARCHAR(255) DEFAULT NULL;';
            $down = 'ALTER TABLE tenancy_tenants '
                .'DROP COLUMN mailer_dsn, DROP COLUMN mailer_from, DROP COLUMN mailer_reply_to;';

            if (!class_exists(\Doctrine\Migrations\AbstractMigration::class)) {
                $io->note('doctrine/migrations not installed. Apply this SQL manually:');
                $io->writeln('  '.$up);
                $io->writeln('-- to revert:');
                $io->writeln('  '.$down);
                return InstallResult::ok('printed SQL snippet — doctrine/migrations absent');
            }

            $timestamp = date('YmdHis');
            $className = 'Version'.$timestamp.'AddTenantMailerColumns';
            $migPath = rtrim($migrationsDir, '/').'/'.$className.'.php';

            $migContent = "<?php\n\ndeclare(strict_types=1);\n\nnamespace DoctrineMigrations;\n\nuse Doctrine\\DBAL\\Schema\\Schema;\nuse Doctrine\\Migrations\\AbstractMigration;\n\nfinal class $className extends AbstractMigration\n{\n    public function getDescription(): string { return 'Add tenant mailer config columns (BOOT-04)'; }\n\n    public function up(Schema \$schema): void\n    {\n        \$this->addSql('$up');\n    }\n\n    public function down(Schema \$schema): void\n    {\n        \$this->addSql('$down');\n    }\n}\n";

            if ($dryRun) {
                $io->section('--with-mailer (dry-run): would write migration');
                $io->text($migPath);
                $io->writeln($migContent);
                return InstallResult::ok('dry-run: would write '.$migPath);
            }

            $this->filesystem->dumpFile($migPath, $migContent);
            $io->success('Wrote migration: '.$migPath);
            return InstallResult::ok('Wrote migration: '.$migPath);
        }

        /**
         * D-09 sub-action 3: append commented-out mailer defaults to config/packages/tenancy.yaml.
         *
         * Behaviour:
         *   - File missing → print snippet to $io, return ok (manual). DO NOT error.
         *   - File present, no top-level `mailer:` key (commented or active) → append block.
         *   - File present, `mailer:` key already present (commented or active) → no-op + $io note.
         *
         * The append is a plain `Filesystem::appendToFile()` — we do NOT parse the YAML
         * with symfony/yaml because we want to preserve user comments / formatting / order.
         * Idempotency is enforced by a regex scan for the literal token "mailer:" in any
         * (commented or uncommented) form at the start of a line.
         */
        private function updateTenancyYaml(string $path, bool $dryRun, SymfonyStyle $io): InstallResult
        {
            if (!is_file($path)) {
                $io->note('config/packages/tenancy.yaml not found at "'.$path.'". Add the following block manually under your tenancy: root config:');
                $io->writeln(self::TENANCY_YAML_MAILER_BLOCK);
                return InstallResult::ok('printed yaml snippet — tenancy.yaml absent at '.$path);
            }

            $contents = (string) file_get_contents($path);

            // Detect any pre-existing mailer key — commented (# mailer:) or active (mailer:).
            // We match at start-of-line (^ multiline) with optional leading whitespace and an
            // optional leading '#'. This catches both commented-out defaults left by a prior
            // --with-mailer run AND a user-uncommented active config.
            if (preg_match('/^[ \t]*#?[ \t]*mailer[ \t]*:/m', $contents) === 1) {
                $io->note('config/packages/tenancy.yaml already contains a `mailer:` section — leaving unchanged.');
                return InstallResult::alreadyInstalled();
            }

            if ($dryRun) {
                $io->section('--with-mailer (dry-run): would append to '.$path);
                $io->writeln(self::TENANCY_YAML_MAILER_BLOCK);
                return InstallResult::ok('dry-run: would append mailer block to '.$path);
            }

            $this->filesystem->appendToFile($path, self::TENANCY_YAML_MAILER_BLOCK."\n");
            $io->success('Appended commented-out `mailer:` defaults to '.$path);
            return InstallResult::ok('Appended mailer block to '.$path);
        }
    }
    ```
