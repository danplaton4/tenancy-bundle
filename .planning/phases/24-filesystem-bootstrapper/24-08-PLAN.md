public function testCacheBoundedUnder100Tenants(): void {
    $cache = self::getContainer()->get('tenancy.mailer.lru_cache');
    for ($i = 0; $i < 100; $i++) {
        $slug = sprintf('tenant_%03d', $i);
        // dispatch a mail with X-Transport: tenant_<slug>
    }
    $this->assertLessThanOrEqual(32, $cache->size());
    $this->assertGreaterThan(0, $cache->evictions());
}
