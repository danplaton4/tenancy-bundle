<?php

declare(strict_types=1);

namespace Tenancy\Bundle\Tests\Integration\Profiler;

use PHPUnit\Framework\TestCase;
use Tenancy\Bundle\Profiler\TenantDataCollector;
use Tenancy\Bundle\Profiler\TenantProfilerStash;
use Tenancy\Bundle\Tests\Integration\TestKernel;

/**
 * Verifies the kernel.debug guard in TenancyBundle::loadExtension() actually compiles out
 * the Profiler services from the production container.
 *
 * Phase 19 — DX-02 acceptance line 6: "service registered ONLY when kernel.debug = true".
 *
 * Both kernels in this test use the same TestKernel class with different (env, debug) pairs.
 * TestKernel::getCacheDir() keys cache by env, so the two kernels never collide.
 */
final class TenantDataCollectorCompileOutTest extends TestCase
{
    public function testCollectorIsRegisteredWhenDebugTrue(): void
    {
        $kernel = new TestKernel('test', true);
        $kernel->boot();

        try {
            self::assertTrue(
                $kernel->getContainer()->has(TenantDataCollector::class),
                'TenantDataCollector MUST be registered when kernel.debug=true'
            );
            self::assertTrue(
                $kernel->getContainer()->has(TenantProfilerStash::class),
                'TenantProfilerStash MUST be registered when kernel.debug=true'
            );
        } finally {
            $kernel->shutdown();
        }
    }

    public function testCollectorIsAbsentWhenDebugFalse(): void
    {
        $kernel = new TestKernel('prod', false);
        $kernel->boot();

        try {
            self::assertFalse(
                $kernel->getContainer()->has(TenantDataCollector::class),
                'TenantDataCollector MUST NOT be registered when kernel.debug=false (compile-out failed)'
            );
            self::assertFalse(
                $kernel->getContainer()->has(TenantProfilerStash::class),
                'TenantProfilerStash MUST NOT be registered when kernel.debug=false (compile-out failed)'
            );
        } finally {
            $kernel->shutdown();
        }
    }

    public function testDataCollectorTagIsPresentWhenDebugTrue(): void
    {
        $kernel = new TestKernel('test', true);
        $kernel->boot();

        try {
            // Find services tagged data_collector with id='tenancy'.
            // Use the container's findTaggedServiceIds via reflection on the test container.
            // Simpler: just resolve the service and assert the tag-attribute side effects.
            self::assertTrue($kernel->getContainer()->has(TenantDataCollector::class));
            $collector = $kernel->getContainer()->get(TenantDataCollector::class);
            self::assertInstanceOf(TenantDataCollector::class, $collector);
            self::assertSame('tenancy', $collector->getName());
            self::assertSame('@Tenancy/Collector/tenant.html.twig', $collector::getTemplate());
        } finally {
            $kernel->shutdown();
        }
    }
}
```

Critical implementation notes:
1. **No `setUpBeforeClass` / `tearDownAfterClass`** — each test boots its own kernel. The two `TestKernel` instances with different `(env, debug)` produce different cache dirs (TestKernel uses `md5(static::class) . '_' . $environment` keying), so they're isolated. Per-test boot/shutdown is cleaner here because each test asserts a distinct kernel mode.
2. **`new TestKernel('test', true)` vs `new TestKernel('prod', false)`** — env name change (`'test'` vs `'prod'`) matters: it changes the cache dir AND it changes which `kernel.environment` value the container sees (also a defence — production code paths read `kernel.environment` for other features).
3. **`try { ... } finally { $kernel->shutdown(); }`** — proper kernel lifecycle; without `shutdown()`, the cache dir leaks across tests.
4. **`get(TenantDataCollector::class)` works because Plan 04 marked the service `->public()`** — if it were private, `get()` would throw `ServiceNotFoundException` (only `has()` works on private services in the test container). Plan 04's `->public()` rationale is verified here.
5. **NO `ReplaceTenancyProviderPass` adjustment needed** — the existing pass (referenced by TestKernel::build) handles `tenancy.provider` for both modes. No new compiler pass.

Potential edge case — if `TestKernel` with `('prod', false)` triggers FrameworkBundle's prod-mode behaviors (e.g. cache pool warmup, secrets vault loading) that fail in a test environment without those resources configured, the test boot may error. Mitigation: the existing TestKernel `framework` config is minimal (`secret/test/http_method_override/handle_all_throwables/php_errors`). <?php

declare(strict_types=1);

namespace Tenancy\Bundle\Tests\Integration\Profiler;

use PHPUnit\Framework\MockObject\MockObject;
use PHPUnit\Framework\TestCase;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Tenancy\Bundle\Context\TenantContext;
use Tenancy\Bundle\Exception\TenantNotFoundException;
use Tenancy\Bundle\Profiler\TenantDataCollector;
use Tenancy\Bundle\Profiler\TenantProfilerStash;
use Tenancy\Bundle\TenantInterface;

/**
 * Verifies TenantDataCollector survives PHP serialize() / unserialize() round-trip.
 *
 * This is the stored-profile reload contract — the Symfony Profiler writes serialized
 * collectors to var/cache/dev/profiler/{token} files and reads them back when the user
 * opens an old profile URL. With scalar-only $this->data (Plan 02 D-11), this is lossless.
 *
 * Phase 19 — DX-02 acceptance line 4.
 */
final class TenantDataCollectorSerializationTest extends TestCase
{
    private TenantProfilerStash&MockObject $stash;
    private TenantContext $tenantContext;

    protected function setUp(): void
    {
        $this->stash = $this->createMock(TenantProfilerStash::class);
        $this->tenantContext = new TenantContext();
    }

    public function testCollectorRoundTripsResolvedState(): void
    {
        $tenant = $this->createMock(TenantInterface::class);
        $tenant->method('getSlug')->willReturn('acme');
        $tenant->method('getName')->willReturn('Acme Corp');
        $this->tenantContext->setTenant($tenant);

        $this->stash->method('getResolvedBy')->willReturn('Tenancy\\Bundle\\Resolver\\HostResolver');
        $this->stash->method('getBootstrapperFqcns')->willReturn(['Tenancy\\Bundle\\Bootstrapper\\Foo', 'Tenancy\\Bundle\\Bootstrapper\\Bar']);
        $this->stash->method('getCapturedException')->willReturn(null);

        $original = new TenantDataCollector($this->stash, $this->tenantContext, 'database_per_tenant', 'default');
        $original->collect(Request::create('/'), new Response());
        $originalData = $original->getData();

        $blob = serialize($original);
        $restored = unserialize($blob);

        self::assertInstanceOf(TenantDataCollector::class, $restored);
        self::assertSame($originalData, $restored->getData(), 'getData() must be byte-identical after serialize/unserialize round-trip');
    }

    public function testCollectorRoundTripsNullState(): void
    {
        $this->stash->method('getResolvedBy')->willReturn(null);
        $this->stash->method('getBootstrapperFqcns')->willReturn([]);
        $this->stash->method('getCapturedException')->willReturn(null);

        $collector = new TenantDataCollector($this->stash, $this->tenantContext, 'shared_db', 'default');
        $collector->collect(Request::create('/'), new Response());

        $restored = unserialize(serialize($collector));

        self::assertInstanceOf(TenantDataCollector::class, $restored);
        self::assertSame($collector->getData(), $restored->getData());
        self::assertSame('null', $restored->getData()['state']);
    }

    public function testCollectorRoundTripsErrorState(): void
    {
        $this->stash->method('getResolvedBy')->willReturn(null);
        $this->stash->method('getBootstrapperFqcns')->willReturn([]);
        $this->stash->method('getCapturedException')->willReturn([
            'class' => TenantNotFoundException::class,
            'message' => 'tenant "ghost" not found',
        ]);

        $collector = new TenantDataCollector($this->stash, $this->tenantContext, 'shared_db', 'default');
        $collector->collect(Request::create('/'), new Response());

        $restored = unserialize(serialize($collector));

        self::assertInstanceOf(TenantDataCollector::class, $restored);
        self::assertSame($collector->getData(), $restored->getData());
        self::assertSame('error', $restored->getData()['state']);
        self::assertSame(TenantNotFoundException::class, $restored->getData()['error']['class']);
        self::assertSame('tenant "ghost" not found', $restored->getData()['error']['message']);
    }

    public function testSerializedBlobContainsNoObjectReferences(): void
    {
        $this->stash->method('getResolvedBy')->willReturn('R');
        $this->stash->method('getBootstrapperFqcns')->willReturn(['A']);
        $this->stash->method('getCapturedException')->willReturn(null);

        $collector = new TenantDataCollector($this->stash, $this->tenantContext, 'database_per_tenant', 'default');
        $collector->collect(Request::create('/'), new Response());

        $blob = serialize($collector);

        // Per DataCollector::__serialize (vendor: symfony/http-kernel), only $this->data round-trips.
        // The blob should NOT contain a serialized stash mock or TenantContext object — those are
        // constructor args, not state. Defensive: the blob must not reference Mock classes or Closure.
        self::assertStringNotContainsString('Closure', $blob);
        self::assertStringNotContainsString('Mock_', $blob);
        self::assertStringNotContainsString('TenantProfilerStash', $blob);
        // The TenantInterface mock (if any) must not survive either.
        self::assertStringNotContainsString('MockObject', $blob);
    }
}
```

<?php

declare(strict_types=1);

namespace Tenancy\Bundle\Tests\Integration\Profiler;

use PHPUnit\Framework\TestCase;

/**
 * Source-layout invariant — defends against accidental "fix" where someone collapses
 * config/services_dev.php into config/services.php "to centralize service registration".
 *
 * Doing so would defeat the kernel.debug compile-out (RESEARCH Pitfall 8), because the
 * profiler service references would then be parsed unconditionally at every container build,
 * regardless of debug mode. The runtime test (TenantDataCollectorCompileOutTest) catches the
 * outcome; this static test catches the cause.
 *
 * No kernel boot — pure file content inspection.
 *
 * Phase 19 — T-19-10 mitigation, complements DX-02 acceptance line 6.
 */
final class SourceLayoutTest extends TestCase
{
    public function testProfilerClassesAreNotReferencedInProductionServicesFile(): void
    {
        $path = __DIR__.'/../../../config/services.php';
        self::assertFileExists($path);
        $contents = file_get_contents($path);
        self::assertIsString($contents);

        self::assertStringNotContainsString(
            'TenantDataCollector',
            $contents,
            'config/services.php MUST NOT reference TenantDataCollector — profiler services live in config/services_dev.php'
        );
        self::assertStringNotContainsString(
            'TenantProfilerStash',
            $contents,
            'config/services.php MUST NOT reference TenantProfilerStash — profiler services live in config/services_dev.php'
        );
        self::assertStringNotContainsString(
            'Tenancy\\Bundle\\Profiler\\',
            $contents,
            'config/services.php MUST NOT reference the Tenancy\\Bundle\\Profiler namespace'
        );
        self::assertStringNotContainsString(
            'services_dev',
            $contents,
            'config/services.php MUST NOT import services_dev — the import happens in TenancyBundle::loadExtension() under a kernel.debug guard'
        );
    }

    public function testProfilerClassesAreReferencedInDevServicesFile(): void
    {
        $path = __DIR__.'/../../../config/services_dev.php';
        self::assertFileExists($path, 'config/services_dev.php MUST exist (created by Plan 04)');
        $contents = file_get_contents($path);
        self::assertIsString($contents);

        self::assertStringContainsString('TenantDataCollector', $contents);
        self::assertStringContainsString('TenantProfilerStash', $contents);
        self::assertStringContainsString("'data_collector'", $contents);
        self::assertStringContainsString("'id' => 'tenancy'", $contents);
        self::assertStringContainsString("'template' => '@Tenancy/Collector/tenant.html.twig'", $contents);
    }

    public function testTenancyBundleGuardsServicesDevImportWithKernelDebug(): void
    {
        $path = __DIR__.'/../../../src/TenancyBundle.php';
        self::assertFileExists($path);
        $contents = file_get_contents($path);
        self::assertIsString($contents);

        self::assertStringContainsString(
            "\$builder->getParameter('kernel.debug')",
            $contents,
            'TenancyBundle MUST guard the services_dev.php import with $builder->getParameter(kernel.debug)'
        );
        self::assertStringContainsString(
            "import('../config/services_dev.php')",
            $contents,
            'TenancyBundle MUST conditionally import config/services_dev.php'
        );
    }
}
